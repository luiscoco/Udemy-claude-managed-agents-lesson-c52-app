// Structured review-panel gate, not a distributed consensus algorithm.
import { z } from "zod";
const Vote = z.strictObject({ agentId: z.string().min(1), roundId: z.string().min(1),
  proposalHash: z.string().regex(/^[a-f0-9]{64}$/), decision: z.enum(["approve", "reject", "abstain"]),
  confidence: z.number().finite().min(0).max(1), reasoning: z.string().trim().min(1).max(2000) });
export type Round = { id: string; proposalHash: string; closesAt: number;
  roster: ReadonlyMap<string, string> }; // Agent ID -> authenticated producer session.
export function tally(round: Round, incoming: readonly { originSessionId: string; raw: unknown }[], now: number) {
  const tally = { approve: 0, reject: 0, abstain: 0 };
  if (!/^[a-f0-9]{64}$/.test(round.proposalHash) || !Number.isFinite(now) ||
      !Number.isFinite(round.closesAt) || round.roster.size !== 3 ||
      new Set(round.roster.values()).size !== 3 || now >= round.closesAt)
    return { passed: false, reason: "Invalid or closed round", tally };
  const seen = new Set<string>();
  try {
    for (const item of incoming) {
      const vote = Vote.parse(item.raw);
      if (round.roster.get(vote.agentId) !== item.originSessionId || seen.has(vote.agentId) ||
          vote.roundId !== round.id || vote.proposalHash !== round.proposalHash)
        throw new Error("Foreign, duplicate or stale vote");
      seen.add(vote.agentId); tally[vote.decision]++;
    }
  } catch { return { passed: false, reason: "Invalid panel evidence", tally }; }
  // This exact policy waits for all three panelists; two approvals and zero rejects.
  const passed = seen.size === round.roster.size && tally.approve >= 2 && tally.reject === 0;
  return { passed, reason: passed ? "Panel rule met" : "Panel rule not met", tally };
}
// Freeze roster/proposal/round in trusted host storage; bind producer sessions
// from authenticated event ingestion. Identities in JSON are not authentication.
// Persist votes uniquely and close/claim the decision atomically in production.
// Confidence is reported metadata, not a calibrated probability or extra vote.
// Independent contexts can still share correlated model/training/prompt errors.
// A panel recommendation is not deployment authorization, empirical correctness,
// fault-tolerant distributed consensus, or a substitute for required human review.
